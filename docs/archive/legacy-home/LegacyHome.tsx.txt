type NominatimPlace = { lat: string; lon: string; display_name: string };
let nominatimQueue: Promise<unknown> = Promise.resolve();
let lastNominatimRequestAt = 0;
const nominatimCache = new Map<string, unknown>();

function queryNominatim(url: string): Promise<unknown> {
  const request = async () => {
    if (nominatimCache.has(url)) return nominatimCache.get(url);
    const wait = Math.max(0, 1000 - (Date.now() - lastNominatimRequestAt));
    if (wait) await new Promise((resolve) => setTimeout(resolve, wait));
    lastNominatimRequestAt = Date.now();
    const response = await fetch(url, {
      headers: { "Accept-Language": "en" },
      referrerPolicy: "strict-origin-when-cross-origin",
    });
    if (!response.ok) throw new Error("The public place search is temporarily unavailable.");
    const result: unknown = await response.json();
    nominatimCache.set(url, result);
    return result;
  };
  const queued = nominatimQueue.then(request, request);
  nominatimQueue = queued.then(() => undefined, () => undefined);
  return queued;
}

export default function Home() {

  const [toast, setToast] = useState("");
  const [error, setError] = useState("");
  const [mobileOpen, setMobileOpen] = useState(false);
  const [selectedProfile, setSelectedProfile] = useState<Profile | null>(null);
  const [chart, setChart] = useState<Record<string, unknown> | null>(null);
  const [palmImage, setPalmImage] = useState<string | null>(null);
  const [palmFile, setPalmFile] = useState<File | null>(null);
  const [tab, setTab] = useState("Overview");
  const [search, setSearch] = useState("");
  const [precision, setPrecision] = useState<"seconds" | "minutes">("seconds");
  const [chartCount, setChartCount] = useState(0);
  const [account, setAccount] = useState<Account | null>(null);
  const [authOpen, setAuthOpen] = useState(false);
  const [authMode, setAuthMode] = useState<"signup" | "signin">("signup");
  const [afterAuth, setAfterAuth] = useState("Dashboard");

  useEffect(() => { if (!toast) return; const id = setTimeout(() => setToast(""), 3600); return () => clearTimeout(id); }, [toast]);
  useEffect(() => () => { if (palmImage) URL.revokeObjectURL(palmImage); }, [palmImage]);

  const loadProfiles = useCallback(async () => {
    try {
      const res = await fetch(`${API}/api/profiles`, { credentials: "include" });
      if (res.status === 401) { setProfiles([]); setChartCount(0); return; }
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.detail || "Could not load saved profiles.");
      }
      setProfiles(await res.json());
      const historyResponse = await fetch(`${API}/api/kundli/history`, { credentials: "include" });
      if (historyResponse.ok) {
        const history: SavedChart[] = await historyResponse.json();
        setChartCount(history.length);
      } else setChartCount(0);
      setError("");
    } catch (err) {
      setChartCount(0);
      setError(err instanceof Error ? err.message : "Could not load saved profiles.");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    try {
      const preferences = localStorage.getItem("jyotish-preferences");
      const value = preferences ? JSON.parse(preferences) : null;
      if (value?.precision === "minutes" || value?.precision === "seconds") {
        setPrecision(value.precision);
      }
    } catch {
      // Keep the default precision when stored preferences are unavailable.
    }
  }, []);

  const restoreSession = useCallback(async () => {
    try {
      const res = await fetch(`${API}/api/auth/me`, { credentials: "include" });
      const data = await res.json();
      setAccount(data.user || null);
      await loadProfiles();
    } catch {
      setProfiles([]);
      setChartCount(0);
    } finally {
      setLoading(false);
    }
  }, [loadProfiles]);

  useEffect(() => { void restoreSession(); }, [restoreSession]);

  function requestAuth(destination: string, mode: "signup" | "signin" = "signup") {
    setAfterAuth(destination);
    setAuthMode(mode);
    setAuthOpen(true);
    setMobileOpen(false);
  }

  function openKundli() {
    if (!account) {
      requestAuth("Kundli Chart");
      return;
    }
    setChart(null);
    setForm(emptyForm);
    setSelectedProfile(null);
    setActive("Kundli Chart");
    setMobileOpen(false);
  }

  function chooseSection(section: string) {
    if (section !== "Dashboard" && !account) {
      requestAuth(section);
      return;
    }
    setActive(section);
    setMobileOpen(false);
    setTab("Overview");
  }

  async function completeAuth(user: Account) {
    setAccount(user);
    setAuthOpen(false);
    setActive(afterAuth);
    setMobileOpen(false);
    if (afterAuth === "Kundli Chart") {
      setChart(null);
      setForm(emptyForm);
      setSelectedProfile(null);
    }
    await loadProfiles();
  }

  async function signOut() {
    try {
      await fetch(`${API}/api/auth/signout`, { method: "POST", credentials: "include" });
    } finally {
      setAccount(null);
      setProfiles([]);
      setChartCount(0);
      setSelectedProfile(null);
      setChart(null);
      setActive("Dashboard");
      setAuthOpen(false);
      setToast("You’re signed out.");
    }
  }

  function update<K extends keyof BirthForm>(key: K, value: BirthForm[K]) {
    setForm((f) => ({ ...f, [key]: value }));
  }

  async function submitBirth(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setBusy(true);
    setError("");

    let latStr = form.latitude.trim();
    let lonStr = form.longitude.trim();
    let birthplace = form.birthplace.trim();

    let numLat = Number(latStr);
    let numLon = Number(lonStr);
    let hasCoords = latStr !== "" && lonStr !== "" && !isNaN(numLat) && !isNaN(numLon) && numLat >= -90 && numLat <= 90 && numLon >= -180 && numLon <= 180;

    // Flexible input resolution:
    // If birthplace is provided without coordinates, attempt to geocode
    if (!hasCoords && birthplace) {
      try {
        if (typeof window !== "undefined" && window.google?.maps?.Geocoder) {
          const geocoder = new window.google.maps.Geocoder();
          const geocodeRes = await new Promise<{ lat: number; lon: number; formatted: string } | null>((resolve) => {
            geocoder.geocode({ address: birthplace }, (results, status) => {
              if (status === "OK" && results?.[0]?.geometry?.location) {
                resolve({
                  lat: results[0].geometry.location.lat(),
                  lon: results[0].geometry.location.lng(),
                  formatted: results[0].formatted_address || birthplace
                });
              } else {
                resolve(null);
              }
            });
          });
          if (geocodeRes) {
            numLat = geocodeRes.lat;
            numLon = geocodeRes.lon;
            latStr = String(numLat);
            lonStr = String(numLon);
            hasCoords = true;
            birthplace = geocodeRes.formatted;
            update("latitude", latStr);
            update("longitude", lonStr);
            update("birthplace", birthplace);
          }
        }
        if (!hasCoords) {
          const data = await queryNominatim(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(birthplace)}&limit=1&addressdetails=1`) as NominatimPlace[];
          if (data.length > 0) {
            numLat = Number(data[0].lat);
            numLon = Number(data[0].lon);
            latStr = String(numLat);
            lonStr = String(numLon);
            hasCoords = true;
            birthplace = data[0].display_name;
            update("latitude", latStr);
            update("longitude", lonStr);
            update("birthplace", birthplace);
          }
        }
      } catch (err) {
        console.warn("Geocode on submit failed:", err);
      }
    }

    if (!hasCoords && !birthplace) {
      setError("Please provide a birthplace search, select a point on the map, or enter latitude and longitude.");
      setBusy(false);
      return;
    }

    if (!hasCoords) {
      setError("Please enter valid numerical latitude (-90 to 90) and longitude (-180 to 180), or select a location on the map / search.");
      setBusy(false);
      return;
    }

    if (!birthplace) {
      birthplace = `Coordinates (${numLat.toFixed(4)}, ${numLon.toFixed(4)})`;
      update("birthplace", birthplace);
    }

    const payload = {
      name: form.name,
      profile_id: selectedProfile?.id || null,
      birth_date: form.birth_date,
      birth_time: form.birth_time,
      birthplace: birthplace,
      latitude: numLat,
      longitude: numLon,
      timezone: form.timezone,
      ayanamsa: form.ayanamsa,
      chart_style: form.chart_style,
      node_type: form.node_type,
      description: form.notes
    };

    try {
      const res = await fetch(`${API}/api/kundli/calculate`, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      });
      const body = await res.json();
      if (!res.ok) throw new Error(body.detail || "Calculation could not be completed.");
      setChart(body);
      if (body.profile) setSelectedProfile(body.profile);
      await loadProfiles();
      setToast("Kundli calculated and saved with exact coordinates.");
    } catch (err) {
      setError(err instanceof Error ? err.message : "The calculation service is unavailable.");
    } finally {
      setBusy(false);
    }
  }

    setSelectedProfile(profile);
    setForm({
      ...emptyForm,
      name: profile.name,
      birth_date: profile.birth_date,
      birth_time: profile.birth_time,

    try {

      e.target.value = "";
      return;

    setPalmImage(URL.createObjectURL(file));

        <div className="mobile-shade" style={{ position: "fixed", inset: 0, background: "#0009", zIndex: 30 }} onClick={() => setMobileOpen(false)}>
          {sidebar}
        </div>

      <main className="main">
        <header className="topbar">
          <div style={{ display: "flex", gap: 12, alignItems: "center" }}>
            <button className="icon-btn mobile-menu" onClick={() => setMobileOpen(true)} aria-label="Open navigation">
              <Menu size={16} />

            </div>

            <button className="icon-btn" aria-label="Search" onClick={() => chooseSection("My Profiles")}>
              <Search size={14} />

            <button className="icon-btn" aria-label="Help" onClick={() => setToast("Birth calculations use exact coordinates, selected sidereal settings, and historical time zones.")}>

            </button>
            <button className="button gold" onClick={openKundli}>
              <Plus size={13} /> New Kundli
            </button>
          </div>

        <div className="content">
          {active === "Dashboard" && <Dashboard profiles={profiles} chartCount={chartCount} loading={loading} onKundli={openKundli} onNavigate={chooseSection} onProfile={addProfile} error={error} />}
          {active === "Kundli Chart" && (
            <Kundli
              form={form}
              update={update}
              advanced={advanced}
              setAdvanced={setAdvanced}
              submit={submitBirth}
              busy={busy}
              error={error}
              chart={chart}
              precision={precision}
              profiles={profiles}
              selectedProfile={selectedProfile}
              onProfile={addProfile}
            />
          )}
          {active === "Palmistry" && <Palmistry image={palmImage} file={palmFile} onImage={onImage} setImage={setPalmImage} toast={setToast} />}
          {active === "Combined Analysis" && <Combined profiles={profiles} onNavigate={chooseSection} />}
          {active === "My Profiles" && <Profiles profiles={filteredProfiles} loading={loading} search={search} setSearch={setSearch} onProfile={addProfile} onNew={openKundli} onDelete={removeProfile} error={error} />}
          {active === "Saved Reports" && <Reports onOpenChart={(report) => {
            const profile = profiles.find((item) => item.id === report.profile_id) || null;
            setSelectedProfile(profile);
            if (profile) {

                ...emptyForm,
                name: profile.name,
                birth_date: profile.birth_date,

                birthplace: profile.birthplace,

                longitude: String(profile.longitude),

                chart_style: String((report.calculation.settings as { chart_style?: string } | undefined)?.chart_style || "north"),

function AuthScreen({ mode, setMode, destination, onComplete, onBack }: { mode: "signup" | "signin"; setMode: (m: "signup" | "signin") => void; destination: string; onComplete: (u: Account) => void; onBack: () => void }) {
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [visible, setVisible] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const signup = mode === "signup";

  async function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError("");
    if (signup && password !== confirmPassword) {
      setError("Those passwords don’t match.");
      return;
    }
    setBusy(true);
    try {
      const response = await fetch(`${API}/api/auth/${signup ? "signup" : "signin"}`, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ username, password })
      });
      const body = await response.json();
      if (!response.ok) {
        const detail = body.detail;
        throw new Error(typeof detail === "string" ? detail : "Please check your username and password.");
      }
      onComplete(body.user);
    } catch (err) {
      setError(err instanceof Error ? err.message : "The account service is unavailable. Please try again.");
    } finally {
      setBusy(false);
    }
  }

  return (
    <main className="auth-screen">
      <div className="auth-nav">
        <button className="brand auth-brand" onClick={onBack}>
          <span className="brand-mark">✳</span>
          <span>
            <strong>Jyotish Studio</strong>
            <small>Clarity in the cosmos</small>
          </span>
        </button>

function LocationPicker({
  value,
  latitude,
  longitude,
  onLocation,
  onName,
  onCoords
}: {
  value: string;
  latitude: string;
  longitude: string;
  onLocation: (name: string, lat: number, lon: number) => void;
  onName: (name: string) => void;
  onCoords: (lat: string, lon: string) => void;
}) {
  const inputRef = useRef<HTMLInputElement>(null);
  const mapContainerRef = useRef<HTMLDivElement>(null);
  const leafletMapRef = useRef<LeafletMap | null>(null);
  const leafletMarkerRef = useRef<LeafletMarker | null>(null);

  // Google Maps refs
  const gmapRef = useRef<HTMLDivElement>(null);
  const gmapObj = useRef<google.maps.Map | null>(null);
  const gmarkerRef = useRef<google.maps.Marker | null>(null);
  const gautocompleteRef = useRef<google.maps.places.Autocomplete | null>(null);
  const ggeocoderRef = useRef<google.maps.Geocoder | null>(null);

  const [engine, setEngine] = useState<"osm" | "google">("osm");
  const [mapKey, setMapKey] = useState("");
  const [customKeyInput, setCustomKeyInput] = useState("");
  const [showKeyConfig, setShowKeyConfig] = useState(false);
  const [googleReady, setGoogleReady] = useState(false);
  const [mapError, setMapError] = useState("");
  const [locating, setLocating] = useState(false);
  const [geocoding, setGeocoding] = useState(false);
  const latestProps = useRef({ value, latitude, longitude, onLocation, onName, onCoords });
  latestProps.current = { value, latitude, longitude, onLocation, onName, onCoords };
  const reverseOsmRef = useRef<(lat: number, lon: number) => Promise<void>>(async () => {});
  const reverseGoogleRef = useRef<(lat: number, lon: number) => void>(() => {});
  const locationInputVersion = useRef(0);

  function coordinatesStillMatch(lat: number, lon: number) {
    return latestProps.current.latitude === lat.toFixed(6)
      && latestProps.current.longitude === lon.toFixed(6);
  }

  // Initialize key from environment or localStorage
  useEffect(() => {
    const envKey = process.env.NEXT_PUBLIC_GOOGLE_MAPS_BROWSER_KEY || "";
    let storedKey = "";
    try {
      storedKey = localStorage.getItem("jyotish_gmaps_browser_key") || "";
    } catch {}
    const active = storedKey.trim() || envKey.trim();
    setMapKey(active);
    setCustomKeyInput(storedKey.trim());
    if (active) {
      setEngine("google");
    } else {
      setEngine("osm");
    }
  }, []);

  // Catch Google Maps authentication failures
  useEffect(() => {
    window.gm_authFailure = () => {
      setMapError("Google Maps key was rejected. Switched to OpenStreetMap.");
      setEngine("osm");
    };
  }, []);

  // Check if google maps is loaded on window
  useEffect(() => {
    if (typeof window !== "undefined" && window.google?.maps?.places) {
      setGoogleReady(true);
    }
  }, [mapKey]);

  // Reverse lookups use a shared queue/cache and are paced for the public API.
  async function reverseGeocodeOsm(lat: number, lon: number) {
    setGeocoding(true);
    try {
      const data = await queryNominatim(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}&zoom=14&addressdetails=1`) as { display_name?: string };
      if (data.display_name && coordinatesStillMatch(lat, lon)) {
        latestProps.current.onName(data.display_name);
      }
    } catch {
      // quiet fallback
    } finally {
      setGeocoding(false);
    }
  }

  // Reverse geocoding helper (Google)
  function reverseGeocodeGoogle(lat: number, lon: number) {
    if (!window.google?.maps) return;
    if (!ggeocoderRef.current) {
      ggeocoderRef.current = new window.google.maps.Geocoder();
    }
    setGeocoding(true);
    ggeocoderRef.current.geocode({ location: { lat, lng: lon } }, (results, status) => {
      setGeocoding(false);
      if (status === "OK" && results?.[0]?.formatted_address && coordinatesStillMatch(lat, lon)) {
        latestProps.current.onName(results[0].formatted_address);
      }
    });
  }

  reverseOsmRef.current = reverseGeocodeOsm;
  reverseGoogleRef.current = reverseGeocodeGoogle;

  // Initialize Built-in Leaflet Map (Zero-Key Mode)
  useEffect(() => {
    if (engine !== "osm") return;
    let isMounted = true;

    async function initLeaflet() {
      if (typeof window === "undefined" || !mapContainerRef.current) return;
      if (leafletMapRef.current) return;

      try {
        const L = (await import("leaflet")).default;
        if (!isMounted || !mapContainerRef.current) return;

        const { latitude: currentLatitude, longitude: currentLongitude } = latestProps.current;
        const numLat = Number(currentLatitude);
        const numLon = Number(currentLongitude);
        const hasInitial = !isNaN(numLat) && !isNaN(numLon) && currentLatitude !== "" && currentLongitude !== "" && numLat >= -90 && numLat <= 90 && numLon >= -180 && numLon <= 180;
        const initialLat = hasInitial ? numLat : 20.5937;
        const initialLon = hasInitial ? numLon : 78.9629;

        const map = L.map(mapContainerRef.current, {
          center: [initialLat, initialLon],
          zoom: hasInitial ? 10 : 3,
          zoomControl: false,
          attributionControl: true,
        });

        // Dark celestial tile layer
        L.tileLayer("https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png", {
          subdomains: "abcd",
          maxZoom: 19,
          attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors &copy; <a href="https://carto.com/attributions">CARTO</a>',
        }).addTo(map);

        L.control.zoom({ position: "bottomright" }).addTo(map);

        const goldIcon = L.divIcon({
          className: "custom-map-pin",
          html: `<div style="transform: translate(-50%, -100%);">
            <svg width="28" height="36" viewBox="0 0 28 36" fill="none" xmlns="http://www.w3.org/2000/svg">
              <path d="M14 0C6.268 0 0 6.268 0 14C0 24.5 14 36 14 36C14 36 28 24.5 28 14C28 6.268 21.732 0 14 0Z" fill="#c8aa6c" stroke="#12130f" stroke-width="1.5"/>
              <circle cx="14" cy="14" r="5" fill="#12130f"/>
              <circle cx="14" cy="14" r="2.5" fill="#f4f0e7"/>
            </svg>
          </div>`,
          iconSize: [28, 36],
          iconAnchor: [14, 36]
        });

        if (hasInitial) {
          const marker = L.marker([initialLat, initialLon], {
            icon: goldIcon,
            draggable: true
          }).addTo(map);

          marker.on("dragend", () => {
            const pos = marker.getLatLng();
            latestProps.current.onCoords(pos.lat.toFixed(6), pos.lng.toFixed(6));
            void reverseOsmRef.current(pos.lat, pos.lng);
          });
          leafletMarkerRef.current = marker;
        }

        map.on("click", (e: LeafletMouseEvent) => {
          const { lat, lng } = e.latlng;
          latestProps.current.onCoords(lat.toFixed(6), lng.toFixed(6));

          if (leafletMarkerRef.current) {
            leafletMarkerRef.current.setLatLng([lat, lng]);
          } else {
            const marker = L.marker([lat, lng], {
              icon: goldIcon,
              draggable: true
            }).addTo(map);

            marker.on("dragend", () => {
              const pos = marker.getLatLng();
              latestProps.current.onCoords(pos.lat.toFixed(6), pos.lng.toFixed(6));
              void reverseOsmRef.current(pos.lat, pos.lng);
            });
            leafletMarkerRef.current = marker;
          }
          void reverseOsmRef.current(lat, lng);
        });

        leafletMapRef.current = map;
      } catch (err) {
        console.warn("Leaflet map init error:", err);
      }
    }

    initLeaflet();
    return () => {
      isMounted = false;
      if (leafletMapRef.current) {
        leafletMapRef.current.remove();
        leafletMapRef.current = null;
        leafletMarkerRef.current = null;
      }
    };
  }, [engine]);

  // Synchronize Leaflet map & marker when manual coordinates change
  useEffect(() => {
    const map = leafletMapRef.current;
    if (engine !== "osm" || !map) return;
    const lat = Number(latitude);
    const lon = Number(longitude);
    if (isNaN(lat) || isNaN(lon) || latitude === "" || longitude === "") return;
    if (lat < -90 || lat > 90 || lon < -180 || lon > 180) return;

    import("leaflet").then((leafletModule) => {
      if (leafletMapRef.current !== map) return;
      const L = leafletModule.default;
      const goldIcon = L.divIcon({
        className: "custom-map-pin",
        html: `<div style="transform: translate(-50%, -100%);">
          <svg width="28" height="36" viewBox="0 0 28 36" fill="none" xmlns="http://www.w3.org/2000/svg">
            <path d="M14 0C6.268 0 0 6.268 0 14C0 24.5 14 36 14 36C14 36 28 24.5 28 14C28 6.268 21.732 0 14 0Z" fill="#c8aa6c" stroke="#12130f" stroke-width="1.5"/>
            <circle cx="14" cy="14" r="5" fill="#12130f"/>
            <circle cx="14" cy="14" r="2.5" fill="#f4f0e7"/>
          </svg>
        </div>`,
        iconSize: [28, 36],
        iconAnchor: [14, 36]
      });

      if (leafletMarkerRef.current) {
        const cur = leafletMarkerRef.current.getLatLng();
        if (Math.abs(cur.lat - lat) > 0.0001 || Math.abs(cur.lng - lon) > 0.0001) {
          leafletMarkerRef.current.setLatLng([lat, lon]);
          map.panTo([lat, lon]);
        }
      } else {
        const marker = L.marker([lat, lon], {
          icon: goldIcon,
          draggable: true
        }).addTo(map);

        marker.on("dragend", () => {
          const pos = marker.getLatLng();
          latestProps.current.onCoords(pos.lat.toFixed(6), pos.lng.toFixed(6));
          void reverseOsmRef.current(pos.lat, pos.lng);
        });
        leafletMarkerRef.current = marker;
        map.panTo([lat, lon]);
      }
    });
  }, [latitude, longitude, engine]);

  // Initialize Google Maps when engine === "google"
  useEffect(() => {
    if (engine !== "google" || !googleReady || !window.google?.maps || !gmapRef.current || gmapObj.current) return;

    try {
      const { latitude: currentLatitude, longitude: currentLongitude, value: currentValue } = latestProps.current;
      const numLat = Number(currentLatitude);
      const numLon = Number(currentLongitude);
      const hasCoords = !isNaN(numLat) && !isNaN(numLon) && currentLatitude !== "" && currentLongitude !== "";
      const centerLat = hasCoords && numLat >= -90 && numLat <= 90 ? numLat : 20.5937;
      const centerLon = hasCoords && numLon >= -180 && numLon <= 180 ? numLon : 78.9629;
      const initialCenter = { lat: centerLat, lng: centerLon };

      const map = new window.google.maps.Map(gmapRef.current, {
        center: initialCenter,
        zoom: hasCoords ? 11 : 3,
        mapTypeControl: false,
        streetViewControl: false,
        fullscreenControl: false,
        zoomControl: true,
        styles: [
          { elementType: "geometry", stylers: [{ color: "#171813" }] },
          { elementType: "labels.text.stroke", stylers: [{ color: "#11120e" }] },
          { elementType: "labels.text.fill", stylers: [{ color: "#9c9a8d" }] },
          { featureType: "administrative.locality", elementType: "labels.text.fill", stylers: [{ color: "#d9b878" }] },
          { featureType: "poi", elementType: "labels.text.fill", stylers: [{ color: "#85867a" }] },
          { featureType: "road", elementType: "geometry", stylers: [{ color: "#252720" }] },
          { featureType: "road.highway", elementType: "geometry", stylers: [{ color: "#3d3624" }] },
          { featureType: "road.highway", elementType: "labels.text.fill", stylers: [{ color: "#c8aa6c" }] },
          { featureType: "water", elementType: "geometry", stylers: [{ color: "#0d1418" }] }
        ]
      });

      ggeocoderRef.current = new window.google.maps.Geocoder();

      if (hasCoords) {
        gmarkerRef.current = new window.google.maps.Marker({
          map,
          position: initialCenter,
          draggable: true,
          title: currentValue || "Birthplace"
        });
        gmarkerRef.current.addListener("dragend", (e: google.maps.MapMouseEvent) => {
          if (!e.latLng) return;
          const lat = e.latLng.lat();
          const lon = e.latLng.lng();
          latestProps.current.onCoords(lat.toFixed(6), lon.toFixed(6));
          reverseGoogleRef.current(lat, lon);
        });
      }

      map.addListener("click", (e: google.maps.MapMouseEvent) => {
        if (!e.latLng) return;
        const lat = e.latLng.lat();
        const lon = e.latLng.lng();
        const point = { lat, lng: lon };
        latestProps.current.onCoords(lat.toFixed(6), lon.toFixed(6));

        if (gmarkerRef.current) {
          gmarkerRef.current.setPosition(point);
        } else {
          gmarkerRef.current = new window.google.maps.Marker({
            map,
            position: point,
            draggable: true,
            title: "Selected Location"
          });
          gmarkerRef.current.addListener("dragend", (ev: google.maps.MapMouseEvent) => {
            if (!ev.latLng) return;
            const dlat = ev.latLng.lat();
            const dlon = ev.latLng.lng();
            latestProps.current.onCoords(dlat.toFixed(6), dlon.toFixed(6));
            reverseGoogleRef.current(dlat, dlon);
          });
        }
        reverseGoogleRef.current(lat, lon);
      });

      gmapObj.current = map;
    } catch (err) {
      console.warn("Google Map init error:", err);
    }
  }, [engine, googleReady]);

  // Google Places Autocomplete
  useEffect(() => {
    if (engine !== "google" || !googleReady || !window.google?.maps?.places || !inputRef.current) return;
    if (gautocompleteRef.current) return;

    try {
      const autocomplete = new window.google.maps.places.Autocomplete(inputRef.current, {
        fields: ["formatted_address", "geometry", "name"]
      });

      autocomplete.addListener("place_changed", () => {
        const place = autocomplete.getPlace();
        if (!place || !place.geometry || !place.geometry.location) return;

        const lat = place.geometry.location.lat();
        const lon = place.geometry.location.lng();
        const placeName = place.formatted_address || place.name || "";
        latestProps.current.onLocation(placeName, lat, lon);

        if (gmapObj.current) {
          const point = { lat, lng: lon };
          gmapObj.current.panTo(point);
          gmapObj.current.setZoom(12);
          if (gmarkerRef.current) {
            gmarkerRef.current.setPosition(point);
          } else {
            gmarkerRef.current = new window.google.maps.Marker({
              map: gmapObj.current,
              position: point,
              draggable: true,
              title: placeName
            });
          }
        }
      });
      gautocompleteRef.current = autocomplete;
    } catch {}
  }, [engine, googleReady]);

  function handleSearchInput(text: string) {
    locationInputVersion.current += 1;
    onName(text);
    // Typed place text must not keep coordinates from a previously selected place.
    onCoords("", "");
  }

  async function searchOsmLocation() {
    const query = value.trim();
    if (query.length < 2) {
      setMapError("Enter at least two characters for the place search.");
      return;
    }
    setGeocoding(true);
    setMapError("");
    const searchVersion = locationInputVersion.current;
    try {
      const data = await queryNominatim(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=1&addressdetails=1`) as NominatimPlace[];
      if (!data.length) {
        setMapError("No matching place was found. Try a nearby town or enter coordinates.");
        return;
      }
      if (searchVersion === locationInputVersion.current) handleSelectSuggestion(data[0]);
    } catch (error) {
      setMapError(error instanceof Error ? error.message : "Place search is temporarily unavailable.");
    } finally {
      setGeocoding(false);
    }
  }

  function handleSelectSuggestion(item: NominatimPlace) {
    const lat = Number(item.lat);
    const lon = Number(item.lon);
    const name = item.display_name;
    onLocation(name, lat, lon);
    if (leafletMapRef.current) {
      leafletMapRef.current.flyTo([lat, lon], 12);
      if (leafletMarkerRef.current) {
        leafletMarkerRef.current.setLatLng([lat, lon]);
      }
    }
  }

  // Locate current user device coordinates
  const handleLocateMe = () => {
    if (!navigator.geolocation) {
      setMapError("Geolocation is not supported by your browser.");
      return;
    }
    setLocating(true);
    setMapError("");
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        setLocating(false);
        const lat = pos.coords.latitude;
        const lon = pos.coords.longitude;
        onCoords(lat.toFixed(6), lon.toFixed(6));

        if (engine === "osm" && leafletMapRef.current) {
          leafletMapRef.current.flyTo([lat, lon], 13);
          if (leafletMarkerRef.current) {
            leafletMarkerRef.current.setLatLng([lat, lon]);
          }
          reverseGeocodeOsm(lat, lon);
        } else if (engine === "google" && gmapObj.current) {
          const point = { lat, lng: lon };
          gmapObj.current.panTo(point);
          gmapObj.current.setZoom(13);
          if (gmarkerRef.current) {
            gmarkerRef.current.setPosition(point);
          }
          reverseGeocodeGoogle(lat, lon);
        }
      },
      (err) => {
        setLocating(false);
        setMapError("Could not retrieve device location: " + err.message);
      },
      { timeout: 10000, enableHighAccuracy: true }
    );
  };

  const handleCenterOnMarker = () => {
    const lat = Number(latitude);
    const lon = Number(longitude);
    if (isNaN(lat) || isNaN(lon)) return;

    if (engine === "osm" && leafletMapRef.current) {
      leafletMapRef.current.flyTo([lat, lon], 12);
    } else if (engine === "google" && gmapObj.current) {
      gmapObj.current.panTo({ lat, lng: lon });
      gmapObj.current.setZoom(12);
    }
  };

  const handleSaveCustomKey = () => {
    const trimmed = customKeyInput.trim();
    try {
      if (trimmed) {
        localStorage.setItem("jyotish_gmaps_browser_key", trimmed);
      } else {
        localStorage.removeItem("jyotish_gmaps_browser_key");
      }
    } catch {}
    setMapKey(trimmed);
    setShowKeyConfig(false);
    setMapError("");
    if (trimmed) {
      setEngine("google");
    } else {
      setEngine("osm");
    }
  };

  const hasCoords = latitude !== "" && longitude !== "" && !isNaN(Number(latitude)) && !isNaN(Number(longitude));

  return (
    <>
      {mapKey && (
        <Script
          id="google-maps-api"
          src={`https://maps.googleapis.com/maps/api/js?key=${encodeURIComponent(mapKey)}&libraries=places,geometry&v=weekly`}
          strategy="afterInteractive"
          onLoad={() => {
            setGoogleReady(true);
            setMapError("");
          }}
          onError={() => {
            setMapError("Google Maps failed to load. Using OpenStreetMap instead.");
            setEngine("osm");
          }}
        />
      )}

      {/* Google Places autocomplete or an explicit single-result public place search. */}
      <div style={{ position: "relative", width: "100%" }}>
        <div style={{ position: "absolute", left: 11, top: 11, color: "#8f7e53", pointerEvents: "none" }}>
          <MapPin size={15} />
        </div>
        <input
          ref={inputRef}
          type="text"
          value={value}
          onChange={(e) => handleSearchInput(e.target.value)}
          placeholder={engine === "google" ? "Search city, town, or address (Google Places)..." : "Enter a city, town, or address, then find it..."}
          style={{ paddingLeft: 34, paddingRight: value ? 60 : 34 }}
          autoComplete="off"
        />
        {value && (
          <button
            type="button"
            aria-label="Clear location"
            onClick={() => {
              locationInputVersion.current += 1;
              onName("");
              onCoords("", "");
              setMapError("");
            }}
            style={{ position: "absolute", right: 10, top: 10, background: "transparent", border: 0, color: "#77766b", cursor: "pointer", padding: 2 }}
          >
            <X size={14} />
          </button>
        )}

      </div>

      {/* Map toolbar & status */}
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginTop: 7, flexWrap: "wrap", gap: 6 }}>
        <span className="field-help" style={{ margin: 0 }}>
          {geocoding ? "Looking up location…" : engine === "google" ? "Google Maps active." : "OpenStreetMap active; no Google key required."}
        </span>
        <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
          {engine === "osm" && (
            <button type="button" className="map-pill-btn" onClick={searchOsmLocation} disabled={geocoding}>
              {geocoding ? "Searching…" : "Find location"}
            </button>
          )}
          <button
            type="button"
            className="map-pill-btn"
            onClick={handleLocateMe}
            disabled={locating}
            title="Use device GPS geolocation"
          >
            <Locate size={11} /> {locating ? "Locating..." : "Locate Me"}
          </button>
          <button
            type="button"
            className="map-pill-btn"
            onClick={() => setShowKeyConfig(!showKeyConfig)}
            title="Google Maps API Key configuration"
          >
            <Key size={11} /> {mapKey ? "Google Key Configured" : "Add Google Maps Key"}
          </button>
        </div>
      </div>
      <p className="field-help">Place search sends the location text to the selected map provider; reverse lookup sends selected coordinates. Birth name, date, and time are not included. Enter coordinates manually for local-only location entry.</p>
      {engine === "osm" && <p className="field-help">OpenStreetMap place search runs only when you select Find location and is paced with a short cache. OpenStreetMap and CARTO attribution appears on the map.</p>}
      {mapError && <div className="notice" role="status">{mapError}</div>}

      {/* Optional key input drawer */}
      {showKeyConfig && (
        <div className="map-key-box">
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
            <span style={{ fontWeight: 500, color: "#d9b878" }}>Google Maps Platform API Key</span>
            <button type="button" onClick={() => setShowKeyConfig(false)} style={{ background: "transparent", border: 0, color: "#888", cursor: "pointer" }}><X size={12} /></button>
          </div>
          <p style={{ margin: "4px 0 0", fontSize: 9, color: "#8a8b80" }}>
            OpenStreetMap works without a key. For Google Maps, obtain a Google Maps demo key for prototyping or configure a restricted browser key, then enter it here or set <code>NEXT_PUBLIC_GOOGLE_MAPS_BROWSER_KEY</code> in <code>.env</code>.
          </p>
          <a href="https://developers.google.com/maps/documentation/javascript/demo-key" target="_blank" rel="noreferrer" style={{ display: "inline-block", marginTop: 6, color: "#d9b878", fontSize: 10 }}>Get a no-cost demo key from Google ↗</a>
          <div style={{ display: "flex", gap: 6, marginTop: 8 }}>
            <input
              type="text"
              value={customKeyInput}
              onChange={(e) => setCustomKeyInput(e.target.value)}
              placeholder="AIzaSy..."
            />
            <button type="button" className="button gold" style={{ padding: "6px 12px", whiteSpace: "nowrap" }} onClick={handleSaveCustomKey}>
              Save Key
            </button>
          </div>
        </div>
      )}

      {/* Interactive Map Canvas */}
      <div className="map-wrapper">
        <div className="map-header">
          <span className="map-title">
            <Globe2 size={12} /> {engine === "google" ? "Google Maps" : "OpenStreetMap"}
          </span>
          <div className="map-actions">
            {mapKey && (
              <button
                type="button"
                className="map-pill-btn"
                onClick={() => setEngine(engine === "google" ? "osm" : "google")}
                title="Toggle between Google Maps and OpenStreetMap"
              >
                Switch to {engine === "google" ? "OpenStreetMap" : "Google Maps"}
              </button>
            )}
            {hasCoords && (
              <button type="button" className="map-pill-btn" onClick={handleCenterOnMarker} title="Center map on current marker">
                <RotateCcw size={10} /> Center Marker
              </button>
            )}
          </div>
        </div>

        {/* Dynamic canvas depending on engine */}
        {engine === "osm" ? (
          <div ref={mapContainerRef} className="map-canvas" aria-label="Celestial Interactive Map" />
        ) : (
          <div ref={gmapRef} className="map-canvas" aria-label="Google Maps" />
        )}

        <div className="map-footer">
          <span>
            {hasCoords ? (
              <>Selected: <span className="map-coords-badge">{Number(latitude).toFixed(6)}°, {Number(longitude).toFixed(6)}°</span></>
            ) : (
              "Click anywhere on the map or drag the marker to set coordinates."
            )}
          </span>
          <span>{geocoding ? "Resolving address..." : "Interactive draggable pin"}</span>
        </div>
      </div>

      {/* Error notice */}
      {mapError && (
        <div className="notice" style={{ marginTop: 8, color: "#e0a59a", borderColor: "#674139" }}>
          {mapError}
        </div>
      )}
    </>
  );
}

function Reports({ onOpenChart }: { onOpenChart: (report: SavedChart) => void }) {
  const [reports, setReports] = useState<SavedChart[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    let active = true;
    fetch(`${API}/api/kundli/history`, { credentials: "include" })
      .then(async (response) => {
        const body = await response.json();
        if (!response.ok) throw new Error(body.detail || "Could not load chart history.");
        if (active) setReports(body);
      })
      .catch((reason) => {
        if (active) setError(reason instanceof Error ? reason.message : "Could not load chart history.");
      })
      .finally(() => { if (active) setLoading(false); });
    return () => { active = false; };
  }, []);

  return (
    <>
      <div className="page-header">
        <p className="eyebrow">Workspace</p>
        <h1 className="heading">Saved Reports</h1>
        <p className="subheading">Review saved Kundli calculations and reopen their chart details.</p>
      </div>
      <div className="card empty-state">
        {loading ? <span>Loading saved chart history…</span> : error ? <><strong>Chart history is unavailable</strong><span>{error}</span></> : reports.length ? (
          <div style={{ width: "100%" }}>
            <div className="empty-icon"><FileText size={20} /></div>
            <strong>{reports.length} saved chart{reports.length === 1 ? "" : "s"}</strong>
            {reports.map((report) => (
              <button className="list-row" key={report.record_id} onClick={() => onOpenChart(report)} style={{ width: "100%", textAlign: "left", cursor: "pointer", color: "inherit", background: "transparent" }}>
                <span><b>{report.profile_name}</b><small>{new Date(report.created_at).toLocaleString()} · {String((report.calculation.ascendant as { sign?: string } | undefined)?.sign || "Ascendant unavailable")}</small></span>
                <ArrowRight size={14} />
              </button>
            ))}
          </div>
        ) : <><div className="empty-icon"><FileText size={20} /></div><strong>No saved charts yet</strong><span>Create a Kundli to see its saved calculation history here.</span></>}
      </div>
    </>
  );
}

function Assistant({ profile, profiles, onSelectProfile }: { profile: Profile | null; profiles: Profile[]; onSelectProfile: (p: Profile | null) => void }) {
  const [question, setQuestion] = useState("");
  const [messages, setMessages] = useState<{ role: string; text: string }[]>([]);
  const [busy, setBusy] = useState(false);
  useEffect(() => { setMessages([]); }, [profile?.id]);

  async function ask(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const q = question.trim();
    if (!q) return;
    const history = messages.slice(-10).map((message) => ({ role: message.role, content: message.text.slice(0, 2000) }));
    setMessages((m) => [...m, { role: "user", text: q }]);
    setQuestion("");
    setBusy(true);
    try {
      const res = await fetch(`${API}/api/assistant`, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ question: q, profile_id: profile?.id ?? null, history })
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || "Assistant is not configured.");
      setMessages((m) => [...m, { role: "assistant", text: data.answer }]);
    } catch (error) {
      const detail = error instanceof Error ? error.message : "The assistant could not answer this request.";
      setMessages((m) => [...m, { role: "assistant", text: detail }]);
    } finally {
      setBusy(false);
    }
  }

  return (
    <>
      <div className="page-header">
        <p className="eyebrow">Research assistant</p>
        <h1 className="heading">AI Astrology Assistant</h1>
        <p className="subheading">Ask about a saved D1 chart or general Jyotish concepts. Answers use the selected calculation and recent chat turns.</p>
      </div>
      <div className="card form-card">
        <div className="field" style={{ maxWidth: 440, marginBottom: 12 }}>
          <label htmlFor="assistant-profile">Chart context</label>
          <select
            id="assistant-profile"
            value={profile?.id ?? ""}
            onChange={(event) => {
              setMessages([]);
              onSelectProfile(profiles.find((item) => item.id === event.target.value) ?? null);
            }}
          >
            <option value="">General question · no chart context</option>
            {profiles.map((item) => <option key={item.id} value={item.id}>{item.name} · {item.birthplace}</option>)}
          </select>
        </div>
        <div className="notice-inline">
          <Sparkles size={13} style={{ verticalAlign: "middle", marginRight: 6, color: "#c7ad70" }} />
          {profile ? `Using saved D1 placements for ${profile.name}.` : "No personal chart data will be sent for this question."} The selected chart’s birth identifiers are omitted. Your question and recent chat turns are sent to Gemini. Classical source citations are unavailable until the reference library is connected.
        </div>
        <div style={{ minHeight: 190, padding: "13px 0" }}>
          {messages.length ? (
            messages.map((m, i) => (
              <div key={i} className="list-row">
                <span>
                  <b>{m.role === "user" ? "You" : "Assistant"}</b>
                  <small style={{ color: "#c4c0b4", lineHeight: 1.6 }}>{m.text}</small>
                </span>
              </div>
            ))
          ) : (
            <div className="empty-state">
              <Sparkles size={20} />
              <strong>Questions with sources, not guesswork</strong>
              <span>Try: “What is a Nakshatra?” or ask about a saved calculation.</span>
            </div>
          )}
        </div>
        <form onSubmit={ask} className="location-box">
          <input value={question} onChange={(e) => setQuestion(e.target.value)} placeholder="Ask a question…" />
          <button className="button gold" disabled={busy || !question.trim()}>
            <ArrowRight size={13} />
          </button>
        </form>
      </div>
    </>
  );
}

function Settings({ precision, setPrecision }: { precision: "seconds" | "minutes"; setPrecision: (value: "seconds" | "minutes") => void }) {
  const [saved, setSaved] = useState(false);
  const [saveError, setSaveError] = useState("");
  const [theme, setTheme] = useState("dark");
  const [gmapsKey, setGmapsKey] = useState("");

  useEffect(() => {
    try {
      const value = localStorage.getItem("jyotish-preferences");
      if (value) {
        const p = JSON.parse(value);
        setTheme(p.theme || "dark");
      }
      const k = localStorage.getItem("jyotish_gmaps_browser_key");
      if (k) setGmapsKey(k);
    } catch {}
  }, []);

  function save() {
    try {
      localStorage.setItem("jyotish-preferences", JSON.stringify({ theme, precision }));
      if (gmapsKey.trim()) {
        localStorage.setItem("jyotish_gmaps_browser_key", gmapsKey.trim());
      } else {
        localStorage.removeItem("jyotish_gmaps_browser_key");
      }
      setSaveError("");
      setSaved(true);
      setTimeout(() => setSaved(false), 2200);
    } catch {
      setSaved(false);
      setSaveError("Browser storage is unavailable, so these settings could not be saved.");
    }
  }

  return (
    <>
      <div className="page-header">
        <p className="eyebrow">Workspace</p>
        <h1 className="heading">Settings</h1>
        <p className="subheading">Personalize birth-time entry, chart defaults, and Google Maps integration on this device.</p>
      </div>
      <div className="card form-card" style={{ maxWidth: 650 }}>
        <div className="field-grid">
          <Field label="Appearance">
            <select value={theme} onChange={(e) => setTheme(e.target.value)}>
              <option value="dark">Dark</option>
              <option value="light" disabled>Light · coming later</option>
            </select>
          </Field>
          <Field label="Birth time precision">
            <select value={precision} onChange={(e) => setPrecision(e.target.value as "seconds" | "minutes")}>
              <option value="seconds">Seconds</option>
              <option value="minutes">Minutes</option>
            </select>
          </Field>
          <div className="field full">
            <label>Google Maps Browser Key (Optional)</label>
            <input
              type="text"
              value={gmapsKey}
              onChange={(e) => setGmapsKey(e.target.value)}
              placeholder="e.g. AIzaSy... (or configure NEXT_PUBLIC_GOOGLE_MAPS_BROWSER_KEY in .env)"
            />
            <span className="field-help">Optional. Enables Google Maps and Google Places search. OpenStreetMap works without a key. <a href="https://developers.google.com/maps/documentation/javascript/demo-key" target="_blank" rel="noreferrer">Get a no-cost demo key ↗</a></span>
          </div>
        </div>
        <button className="button gold" style={{ marginTop: 17 }} onClick={save}>
          <Save size={13} />
          {saved ? "Preferences saved" : "Save preferences"}
        </button>
        {saveError && <div className="notice" role="alert">{saveError}</div>}
        <div className="notice">Preferences and API keys saved here remain in your browser. Server credentials are never sent to the browser.</div>
      </div>
    </>
  );
}
